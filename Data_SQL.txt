-- ===================================================================
-- 1. ตารางข้อมูลหลัก (Master / Entity Tables)
-- ===================================================================

-- 1.1 ตารางผู้ใช้งาน (Users)
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(150) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    target_career VARCHAR(100), -- อาชีพเป้าหมายที่ระบุ
    role VARCHAR(20) DEFAULT 'user' CHECK (role IN ('user', 'admin')),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 1.2 ตารางคอร์สเรียน (Courses)
CREATE TABLE courses (
    id SERIAL PRIMARY KEY,
    title VARCHAR(150) NOT NULL,
    description TEXT,
    thumbnail_url VARCHAR(255),
    price DECIMAL(10, 2) DEFAULT 0.00,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 1.3 ตารางบริษัท (Companies)
CREATE TABLE companies (
    id SERIAL PRIMARY KEY,
    name VARCHAR(150) NOT NULL,
    logo_url VARCHAR(255),
    description TEXT,
    location VARCHAR(150),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 1.4 ตารางทักษะ/ความสามารถ (Skills)
CREATE TABLE skills (
    id SERIAL PRIMARY KEY,
    skill_name VARCHAR(100) UNIQUE NOT NULL,
    category VARCHAR(50) -- เช่น 'Hard Skill', 'Soft Skill', 'Language'
);

-- 1.5 ตารางประกาศงาน (Jobs)
CREATE TABLE jobs (
    id SERIAL PRIMARY KEY,
    company_id INT REFERENCES companies(id) ON DELETE CASCADE,
    title VARCHAR(150) NOT NULL,
    description TEXT,
    min_salary DECIMAL(10, 2),
    max_salary DECIMAL(10, 2),
    is_high_income BOOLEAN DEFAULT FALSE, -- ธงระบุว่าเป็นอาชีพรายได้สูงสำหรับนำไปจัดอันดับหน้าแรก
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ===================================================================
-- 2. ตารางเชื่อมโยงความสัมพันธ์ และประวัติ (Mapping & History Tables)
-- ===================================================================

-- 2.1 ตารางประวัติการเรียนของผู้ใช้ (User Courses)
CREATE TABLE user_courses (
    id SERIAL PRIMARY KEY,
    user_id INT REFERENCES users(id) ON DELETE CASCADE,
    course_id INT REFERENCES courses(id) ON DELETE CASCADE,
    progress_percent INT DEFAULT 0 CHECK (progress_percent BETWEEN 0 AND 100),
    status VARCHAR(20) DEFAULT 'enrolled' CHECK (status IN ('enrolled', 'completed', 'dropped')),
    enrolled_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    completed_at TIMESTAMP,
    UNIQUE(user_id, course_id)
);

-- 2.2 ตารางทักษะที่ผู้ใช้มี (User Skills)
CREATE TABLE user_skills (
    user_id INT REFERENCES users(id) ON DELETE CASCADE,
    skill_id INT REFERENCES skills(id) ON DELETE CASCADE,
    level INT DEFAULT 1 CHECK (level BETWEEN 1 AND 5), -- ระดับ 1-5
    PRIMARY KEY (user_id, skill_id)
);

-- 2.3 ตารางทักษะที่จะได้รับจากคอร์สเรียน (Course Skills)
CREATE TABLE course_skills (
    course_id INT REFERENCES courses(id) ON DELETE CASCADE,
    skill_id INT REFERENCES skills(id) ON DELETE CASCADE,
    skill_boost_level INT DEFAULT 1, -- เรียนจบแล้วเพิ่มระดับทักษะเท่าไหร่
    PRIMARY KEY (course_id, skill_id)
);

-- 2.4 ตารางทักษะที่งานต้องการ (Job Skills - สำหรับใช้เปรียบเทียบตลาดงาน)
CREATE TABLE job_skills (
    job_id INT REFERENCES jobs(id) ON DELETE CASCADE,
    skill_id INT REFERENCES skills(id) ON DELETE CASCADE,
    required_level INT DEFAULT 1 CHECK (required_level BETWEEN 1 AND 5),
    PRIMARY KEY (job_id, skill_id)
);

-- 2.5 ตารางประวัติการแชทกับ AI (Chatbot Logs)
CREATE TABLE chatbot_logs (
    id SERIAL PRIMARY KEY,
    user_id INT REFERENCES users(id) ON DELETE CASCADE,
    message TEXT NOT NULL,
    is_ai_reply BOOLEAN DEFAULT FALSE, -- TRUE = คำตอบจาก AI, FALSE = คำถามจาก User
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
